cmake_minimum_required(VERSION 3.20)
project(CraftPP CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Release = -O2 (on a Pi -O3 only grows the binary; docs/performance.md). Worldgen must stay bit-exact with
# vanilla: never -ffast-math, and -ffp-contract=off (GCC defaults to fused multiply-add contraction).
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
add_compile_options(-ffp-contract=off)
# Sanitizer builds (./build.sh asan|tsan): address = ASan + UBSan, thread = TSan.
set(CRAFTPP_SANITIZE "" CACHE STRING "empty, address or thread")
if(CRAFTPP_SANITIZE STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
elseif(CRAFTPP_SANITIZE STREQUAL "thread")
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=thread)
endif()
set(CRAFTPP_CPU "" CACHE STRING "Tune for one CPU, e.g. cortex-a72 (Pi 4) or cortex-a76 (Pi 5); empty = portable")
if(CRAFTPP_CPU)
  add_compile_options(-mcpu=${CRAFTPP_CPU})
endif()
option(CRAFTPP_LTO "Link-time optimization" OFF)
if(CRAFTPP_LTO)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()
file(GLOB CRAFTPP_MOB_SOURCES CONFIGURE_DEPENDS src/mobs/*.cpp src/mobs/*/*.cpp src/effects/*.cpp src/progress/*.cpp)
add_executable(craftpp src/main.cpp src/boss/end_worldgen.cpp src/light/light.cpp src/storage/world_io.cpp src/storage/level.cpp ${CRAFTPP_MOB_SOURCES})
# Game code is a Java port: signed int overflow wraps like Java instead of being UB.
target_compile_options(craftpp PRIVATE -Wall -Wextra -fwrapv)
# logs/latest.log is gzipped on rollover when the system zlib is available (else kept as plain .log).
find_package(ZLIB QUIET)
if(ZLIB_FOUND)
  target_compile_definitions(craftpp PRIVATE CRAFTPP_ZLIB=1)
  target_link_libraries(craftpp PRIVATE ZLIB::ZLIB)
endif()
add_executable(worldgen_tests
  test/biome_test.cpp src/worldgen/biome_source.cpp)
target_include_directories(worldgen_tests PRIVATE src)
target_compile_options(worldgen_tests PRIVATE -Wall -Wextra)
# Worldgen foundation: RNG, noise, JSON (bit-exact with vanilla; never use -ffast-math here).
add_library(craftpp_worldgen STATIC src/worldgen/noise.cpp src/worldgen/biome_source.cpp
  src/worldgen/density.cpp src/worldgen/terrain.cpp src/worldgen/block_state.cpp src/worldgen/pipeline.cpp
  src/worldgen/carver.cpp src/worldgen/overworld.cpp src/worldgen/features.cpp
  src/worldgen/block_behavior.cpp src/worldgen/feature_basic.cpp src/worldgen/feature_tree.cpp src/worldgen/feature_cave.cpp
  src/worldgen/feature_misc.cpp src/worldgen/feature_nether_end.cpp src/storage/region.cpp src/storage/chunk_nbt.cpp)
target_include_directories(craftpp_worldgen PUBLIC src)
target_compile_options(craftpp_worldgen PRIVATE -Wall -Wextra)
find_package(ZLIB REQUIRED)  # chunk files (world.hpp)
find_package(Threads REQUIRED)
target_link_libraries(craftpp_worldgen PUBLIC ZLIB::ZLIB Threads::Threads)
target_link_libraries(craftpp PRIVATE craftpp_worldgen)
# Test targets are skipped until their source exists (keeps in-progress work from breaking the build).
if(EXISTS ${CMAKE_SOURCE_DIR}/test/worldgen_rng_test.cpp)
  add_executable(worldgen_rng_test test/worldgen_rng_test.cpp)
  target_link_libraries(worldgen_rng_test PRIVATE craftpp_worldgen)
  target_compile_options(worldgen_rng_test PRIVATE -Wall -Wextra)
endif()
# Material rules stage (surface): part of craftpp_worldgen.
if(EXISTS ${CMAKE_SOURCE_DIR}/src/worldgen/material.cpp)
  target_sources(craftpp_worldgen PRIVATE src/worldgen/material.cpp)
endif()
if(EXISTS ${CMAKE_SOURCE_DIR}/test/material_test.cpp)
  add_executable(material_test test/material_test.cpp)
  target_link_libraries(material_test PRIVATE craftpp_worldgen)
  target_compile_options(material_test PRIVATE -Wall -Wextra)
endif()
if(EXISTS ${CMAKE_SOURCE_DIR}/test/terrain_test.cpp)
  add_executable(terrain_test test/terrain_test.cpp)
  target_link_libraries(terrain_test PRIVATE craftpp_worldgen)
  target_compile_options(terrain_test PRIVATE -Wall -Wextra)
endif()
# Bukkit/Paper plugins (src/plugins, docs/plugins.md). libjvm is dlopen'd at runtime only when plugins/*.jar
# exist, so nothing links against Java; jni.h is needed at build time (any JDK), else plugins are compiled out.
file(GLOB CRAFTPP_PLUGIN_SOURCES CONFIGURE_DEPENDS src/plugins/*.cpp)
target_sources(craftpp PRIVATE ${CRAFTPP_PLUGIN_SOURCES})
set(CRAFTPP_JNI_HOME "$ENV{JAVA_HOME}")
if(NOT EXISTS "${CRAFTPP_JNI_HOME}/include/jni.h")
  file(GLOB CRAFTPP_JDKS /Library/Java/JavaVirtualMachines/*/Contents/Home /usr/lib/jvm/*)
  foreach(jdk ${CRAFTPP_JDKS})
    if(EXISTS "${jdk}/include/jni.h")
      set(CRAFTPP_JNI_HOME "${jdk}")
    endif()
  endforeach()
endif()
if(EXISTS "${CRAFTPP_JNI_HOME}/include/jni.h")
  file(GLOB CRAFTPP_JNI_MD "${CRAFTPP_JNI_HOME}/include/*/jni_md.h")
  get_filename_component(CRAFTPP_JNI_MD_DIR "${CRAFTPP_JNI_MD}" DIRECTORY)
  set_source_files_properties(src/plugins/plugins.cpp PROPERTIES
    COMPILE_DEFINITIONS CRAFTPP_JNI=1 INCLUDE_DIRECTORIES "${CRAFTPP_JNI_HOME}/include;${CRAFTPP_JNI_MD_DIR}")
  target_link_libraries(craftpp PRIVATE ${CMAKE_DL_LIBS})
else()
  message(STATUS "No JDK headers found: building without plugin support")
endif()
# Boss fights (src/boss): pure parts (End spikes, podium, gateways, Mth tables).
add_executable(boss_test test/boss_test.cpp)
target_link_libraries(boss_test PRIVATE craftpp_worldgen)
target_compile_options(boss_test PRIVATE -Wall -Wextra -UNDEBUG)
# Structures (src/worldgen/structure): placement, starts, jigsaw, legacy pieces, beardifier, /locate.
find_package(ZLIB REQUIRED)
target_sources(craftpp_worldgen PRIVATE src/worldgen/structure/blocks.cpp src/worldgen/structure/nbt_io.cpp
  src/worldgen/structure/template.cpp src/worldgen/structure/structure.cpp src/worldgen/structure/jigsaw.cpp
  src/worldgen/structure/structures.cpp src/worldgen/structure/scattered.cpp src/worldgen/structure/templated.cpp
  src/worldgen/structure/beardifier.cpp src/worldgen/structure/hooks.cpp src/worldgen/structure/fortress.cpp src/worldgen/structure/endcity.cpp src/worldgen/structure/mineshaft.cpp src/worldgen/structure/stronghold.cpp src/worldgen/structure/monument.cpp src/worldgen/structure/mansion.cpp)
target_link_libraries(craftpp_worldgen PUBLIC ZLIB::ZLIB)
add_executable(structure_test test/structure_test.cpp)
target_link_libraries(structure_test PRIVATE craftpp_worldgen)
target_compile_options(structure_test PRIVATE -Wall -Wextra)
add_executable(pipeline_test test/pipeline_test.cpp)
target_link_libraries(pipeline_test PRIVATE craftpp_worldgen)
# Network security (src/net): vendored crypto for online mode / signed chat; OpenSSL only for the HTTPS client.
add_library(craftpp_net STATIC src/net/crypto.cpp src/net/http.cpp)
target_include_directories(craftpp_net PUBLIC src)
target_compile_options(craftpp_net PRIVATE -Wall -Wextra)
option(CRAFTPP_TLS "HTTPS to Mojang's session server (online-mode) via the system OpenSSL" ON)
if(CRAFTPP_TLS)
  if(APPLE AND NOT OPENSSL_ROOT_DIR AND EXISTS /opt/homebrew/opt/openssl@3)
    set(OPENSSL_ROOT_DIR /opt/homebrew/opt/openssl@3)
  endif()
  find_package(OpenSSL QUIET)
endif()
if(OpenSSL_FOUND)
  target_compile_definitions(craftpp_net PRIVATE CRAFTPP_TLS=1)
  target_link_libraries(craftpp_net PRIVATE OpenSSL::SSL)
else()
  message(STATUS "No OpenSSL: online-mode cannot reach https://sessionserver.mojang.com (install libssl-dev)")
endif()
target_link_libraries(craftpp PRIVATE craftpp_net)
add_executable(crypto_test test/crypto_test.cpp)
target_link_libraries(crypto_test PRIVATE craftpp_net)
# Mob roster checks: trading / breeding / taming logic (runs from the repo root, needs data/*.json).
add_executable(roster_test test/roster_test.cpp ${CRAFTPP_PLUGIN_SOURCES} src/light/light.cpp src/storage/world_io.cpp src/storage/level.cpp ${CRAFTPP_MOB_SOURCES})
target_include_directories(roster_test PRIVATE src)
target_link_libraries(roster_test PRIVATE craftpp_worldgen craftpp_net ${CMAKE_DL_LIBS})
